Usage: codeintel build schema compile [ARGS]

Compile a deterministic schema manifest for selected targets.

╭─ Parameters ─────────────────────────────────────────────────────────────────╮
│ TARGETS --targets            Target names to include (defaults to all        │
│   --empty-targets            targets). [env var:                             │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_TARGETS]         │
│                              [default: (none)]                               │
│ MODULE --module -m           Compile schemas for all targets in a module.    │
│                              [env var:                                       │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_MODULE]          │
│                              [default: (none)]                               │
│ ALL --all -a                 Compile schemas for all targets across all      │
│                              modules. [env var:                              │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_ALL] [default:   │
│                              false]                                          │
│ ONLY-NATIVE --only-native    Restrict compilation to targets with native     │
│                              Hamilton implementations. [env var:             │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_ONLY_NATIVE]     │
│                              [default: false]                                │
│ INFER-NATIVE --infer-native  Infer schemas for inferable native targets      │
│   --infer                    (fallback to declared on errors). [env var:     │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_INFER_NATIVE]    │
│                              [default: true]                                 │
│ STABLE --stable              Force deterministic ordering and canonicalized  │
│                              output. [env var:                               │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_STABLE]          │
│                              [default: true]                                 │
│ FORMAT --format -f           Output format: json (default). [env var:        │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_FORMAT]          │
│                              [default: json]                                 │
│ OUTPUT --output -o           Output file path (stdout if not specified).     │
│                              [env var:                                       │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_OUTPUT]          │
│                              [default: (none)]                               │
│ INCLUDE-VIEWS                Include DuckDB view schemas in the manifest (v2 │
│   --include-views            format). [env var:                              │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_INCLUDE_VIEWS]   │
│                              [default: false]                                │
│ INCLUDE-ARTIFACTS            Include export artifact metadata in the         │
│   --include-artifacts        manifest (v2 format). [env var:                 │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_INCLUDE_ARTIFACT │
│                              S] [default: false]                             │
│ --flags                      [env var: CODEINTEL_BUILD_SCHEMA_COMPILE_FLAGS] │
│                              [default: --flags]                              │
│ ROOT --root -r               Explicit project root directory. [env var:      │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_ROOT] [default:  │
│                              (none)]                                         │
│ OUTPUT-FORMAT                Output format. [choices: text, json, jsonl]     │
│   --output-format            [env var:                                       │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_OUTPUT_FORMAT]   │
│                              [default: text]                                 │
│ JSON --json                  Alias for --output-format json. [env var:       │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_JSON] [default:  │
│                              false]                                          │
│ VERBOSE --verbose -v         Increase verbosity (0=warnings, 1=info,         │
│                              2=debug). [env var:                             │
│                              CODEINTEL_BUILD_SCHEMA_COMPILE_VERBOSE]         │
│                              [default: 0]                                    │
╰──────────────────────────────────────────────────────────────────────────────╯
